cmake_minimum_required(VERSION 3.24)
project(SouiRenderPhaseCostContracts LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(RenderPhaseCostContracts fixture.cpp)
target_link_libraries(RenderPhaseCostContracts PRIVATE Threads::Threads)
get_filename_component(SOUI_PHASE_FIXTURE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
target_include_directories(RenderPhaseCostContracts PRIVATE "${SOUI_PHASE_FIXTURE_ROOT}")
set_target_properties(RenderPhaseCostContracts PROPERTIES
  CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(RenderPhaseCostContracts PRIVATE /W4 /EHsc /utf-8 /U_DEBUG)
  set_property(TARGET RenderPhaseCostContracts PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(RenderPhaseCostContracts PRIVATE /clang:-fno-elide-constructors)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(RenderPhaseCostContracts PRIVATE -fno-elide-constructors)
endif()
enable_testing()
add_test(NAME benchmark.render-phase-cost COMMAND RenderPhaseCostContracts)
set_tests_properties(benchmark.render-phase-cost PROPERTIES TIMEOUT 10)
