cmake_minimum_required(VERSION 3.28)
project(SouiReferenceAgentRoot LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR
   NOT CMAKE_GENERATOR_TOOLSET STREQUAL "v143,host=x64,version=14.44.35207" OR
   NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL "10.0.19041.0")
  message(FATAL_ERROR "Use fixed MSVC14.44.35207 / exact SDK10.0.19041.0")
endif()
if(CMAKE_VS_PLATFORM_NAME STREQUAL "x64")
  set(reference_cl_arch "x64")
elseif(CMAKE_VS_PLATFORM_NAME STREQUAL "Win32")
  set(reference_cl_arch "x86")
else()
  message(FATAL_ERROR "Only pinned x64 / Win32 compiler targets are supported")
endif()
file(TO_CMAKE_PATH "${CMAKE_CXX_COMPILER}" reference_actual_cl)
set(reference_expected_cl "${CMAKE_GENERATOR_INSTANCE}/VC/Tools/MSVC/14.44.35207/bin/Hostx64/${reference_cl_arch}/cl.exe")
cmake_path(NORMAL_PATH reference_actual_cl)
cmake_path(NORMAL_PATH reference_expected_cl)
string(TOLOWER "${reference_actual_cl}" reference_actual_cl)
string(TOLOWER "${reference_expected_cl}" reference_expected_cl)
if(NOT reference_actual_cl STREQUAL reference_expected_cl)
  message(FATAL_ERROR "Actual CXX compiler differs from the pinned MSVC14.44.35207 path")
endif()
if(NOT IS_ABSOLUTE "${REFERENCE_ROOT}" OR NOT IS_ABSOLUTE "${REFERENCE_NODE}")
  message(FATAL_ERROR "Root must pass the pinned private source and Node")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS ActionJson)
if(NOT TARGET Soui::Sdk OR NOT TARGET Soui::ActionJson)
  message(FATAL_ERROR "Actual installed public targets are required")
endif()
add_custom_target(reference_actions_check
  COMMAND "${REFERENCE_NODE}" "F:/work/sm05-broker/tools/sdk/actiongen.cjs"
    --source "${REFERENCE_ROOT}/contracts/actions.json" --root "${REFERENCE_ROOT}" --check VERBATIM)
add_library(reference_business STATIC "${REFERENCE_ROOT}/app/reference-business.cpp")
target_link_libraries(reference_business PUBLIC Soui::Sdk Soui::ActionJson)
target_compile_features(reference_business PUBLIC cxx_std_17)
target_compile_options(reference_business PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
set_property(TARGET reference_business PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_include_directories(reference_business PUBLIC "${REFERENCE_ROOT}/app")
add_dependencies(reference_business reference_actions_check)
add_library(reference_agent STATIC "${REFERENCE_ROOT}/app/reference-agent.cpp")
target_link_libraries(reference_agent PUBLIC reference_business PRIVATE advapi32)
target_compile_features(reference_agent PUBLIC cxx_std_17)
target_compile_options(reference_agent PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
set_property(TARGET reference_agent PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(soui_reference_agent_tests "${REFERENCE_ROOT}/tests/native-agent.cpp")
target_link_libraries(soui_reference_agent_tests PRIVATE reference_agent user32)
target_link_options(soui_reference_agent_tests PRIVATE /MAP:${CMAKE_BINARY_DIR}/reference-agent-$<CONFIG>.map)
target_compile_features(soui_reference_agent_tests PRIVATE cxx_std_17)
target_compile_options(soui_reference_agent_tests PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
set_property(TARGET soui_reference_agent_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
enable_testing()
foreach(name IN ITEMS disabled-default-deny selected-peer fixed-summary cancel invalid-form revoked-document close-pending page-isolation)
  add_test(NAME reference.agent.${name} COMMAND soui_reference_agent_tests "${name}" "${REFERENCE_ROOT}/common/fixture.json")
  set_tests_properties(reference.agent.${name} PROPERTIES TIMEOUT 20)
endforeach()
get_property(actual_tests DIRECTORY PROPERTY TESTS)
list(SORT actual_tests)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/actual-registered-tests.txt" CONTENT "$<JOIN:${actual_tests},\n>\n" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/actual-sdk-location-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:Soui::Sdk>\n" NEWLINE_STYLE UNIX)
