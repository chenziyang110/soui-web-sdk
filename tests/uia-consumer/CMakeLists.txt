cmake_minimum_required(VERSION 3.24)
project(SouiInstalledUiaConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This installed UIA consumer requires Windows/MSVC")
endif()
set(SOUI_CONSUMER_PREFIX "" CACHE PATH "Exact private installed SDK prefix under test")
if(NOT IS_ABSOLUTE "${SOUI_CONSUMER_PREFIX}" OR NOT EXISTS "${SOUI_CONSUMER_PREFIX}/lib/cmake/SouiSdk/SouiSdkConfig.cmake")
  message(FATAL_ERROR "Select the actual installed or relocated SDK prefix")
endif()
set(SouiSdk_DIR "${SOUI_CONSUMER_PREFIX}/lib/cmake/SouiSdk" CACHE PATH "Explicit installed config" FORCE)
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS Uia PATHS "${SouiSdk_DIR}" NO_DEFAULT_PATH)
if(NOT SouiSdk_Uia_FOUND OR NOT TARGET Soui::Uia)
  message(FATAL_ERROR "The installed SDK must explicitly provide its Uia component")
endif()
get_target_property(soui_uia_imported Soui::Uia IMPORTED)
if(NOT soui_uia_imported)
  message(FATAL_ERROR "Consume the installed package, not a source/alias target")
endif()
add_executable(soui_installed_uia_consumer main.cpp)
target_link_libraries(soui_installed_uia_consumer PRIVATE Soui::Uia)
target_compile_features(soui_installed_uia_consumer PRIVATE cxx_std_17)
target_compile_options(soui_installed_uia_consumer PRIVATE /EHsc /utf-8 /W4 /U_DEBUG)
set_property(TARGET soui_installed_uia_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME installed.uia.parameter-rejection COMMAND soui_installed_uia_consumer)
  set_tests_properties(installed.uia.parameter-rejection PROPERTIES TIMEOUT 10)
endif()
