cmake_minimum_required(VERSION 3.24)
project(SouiSdkPerformanceHost LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This consumer requires the matching Windows/MSVC SDK")
endif()
if(SOUI_NATIVE_TOOLCHAIN STREQUAL "LLVM22MSVC142")
  if(NOT CMAKE_GENERATOR STREQUAL "Ninja Multi-Config" OR NOT COMMAND soui_llvm22_verify_project
      OR NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.19041.0")
    message(FATAL_ERROR "Use the explicit product LLVM22MSVC142 toolchain with Windows SDK 10.0.19041.0")
  endif()
  soui_llvm22_verify_project()
elseif(NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL "10.0.19041.0")
  message(FATAL_ERROR "Freeze Windows SDK 10.0.19041.0 before configuring this consumer")
endif()
set(MATCHED_SDK_ROOT "" CACHE PATH "Frozen independently installed SDK root")
set(MATCHED_SDK_VERSION "" CACHE STRING "Exact installed SDK numeric package version")
set(SOUI_PERF_BUILD_IDENTITY "" CACHE STRING "Frozen baseline/candidate build identity")
option(SOUI_PERF_OPTIONAL_FRAME_FEEDBACK "Compile new optional feedback virtual overrides" OFF)
if(NOT MATCHED_SDK_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Set an explicit MATCHED_SDK_VERSION; EXACT is required")
endif()
string(LENGTH "${SOUI_PERF_BUILD_IDENTITY}" _identity_length)
if(NOT SOUI_PERF_BUILD_IDENTITY MATCHES "^[A-Za-z0-9_.-]+$" OR _identity_length GREATER 80 OR NOT MATCHED_SDK_ROOT)
  message(FATAL_ERROR "Set frozen MATCHED_SDK_ROOT and an ASCII build identity")
endif()
find_package(SouiSdk ${MATCHED_SDK_VERSION} EXACT CONFIG REQUIRED COMPONENTS WebView2
  PATHS "${MATCHED_SDK_ROOT}/lib/cmake/SouiSdk" NO_DEFAULT_PATH)
get_filename_component(_requested_sdk "${MATCHED_SDK_ROOT}" REALPATH)
get_filename_component(_resolved_sdk "${SouiSdk_DIR}/../../.." REALPATH)
if(NOT _requested_sdk STREQUAL _resolved_sdk)
  message(FATAL_ERROR "Resolved SDK differs from explicitly frozen root")
endif()
add_executable(SouiSdkPerformanceHost main.cpp)
target_link_libraries(SouiSdkPerformanceHost PRIVATE Soui::WebView2 Soui::ActionJson bcrypt psapi)
target_compile_features(SouiSdkPerformanceHost PRIVATE cxx_std_17)
target_compile_options(SouiSdkPerformanceHost PRIVATE /EHsc /W4 /utf-8 /U_DEBUG)
target_compile_definitions(SouiSdkPerformanceHost PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601
  SOUI_PERF_PACKAGE_VERSION="${MATCHED_SDK_VERSION}" SOUI_PERF_BUILD_IDENTITY="${SOUI_PERF_BUILD_IDENTITY}"
  SOUI_PERF_OPTIONAL_FRAME_FEEDBACK=$<BOOL:${SOUI_PERF_OPTIONAL_FRAME_FEEDBACK}>)
set_property(TARGET SouiSdkPerformanceHost PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_custom_command(TARGET SouiSdkPerformanceHost POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources" "$<TARGET_FILE_DIR:SouiSdkPerformanceHost>/resources"
  VERBATIM)
