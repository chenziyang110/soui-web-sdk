cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiHostFocusProgress LANGUAGES CXX)
  enable_testing()
endif()
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Mounted focus progress requires Windows/MSVC-compatible SDK")
endif()
get_filename_component(focus_sdk_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET Soui::Sdk)
  if(NOT EXISTS "${MATCHED_SDK_LIBRARY}")
    message(FATAL_ERROR "Standalone build requires the current MATCHED_SDK_LIBRARY")
  endif()
  add_library(SouiFocusMatchedSdk STATIC IMPORTED)
  set_target_properties(SouiFocusMatchedSdk PROPERTIES IMPORTED_LOCATION "${MATCHED_SDK_LIBRARY}")
  set(focus_sdk_target SouiFocusMatchedSdk)
else()
  set(focus_sdk_target Soui::Sdk)
endif()
add_executable(soui_host_focus_progress fixture.cpp)
target_include_directories(soui_host_focus_progress PRIVATE "${focus_sdk_root}/sdk/include"
  "${focus_sdk_root}/sdk/src/host" "${focus_sdk_root}/third_party/include/json")
target_link_libraries(soui_host_focus_progress PRIVATE ${focus_sdk_target}
  user32 gdi32 imm32 shell32 ole32 oleaut32 uuid comctl32 advapi32 winmm bcrypt dbghelp)
target_compile_features(soui_host_focus_progress PRIVATE cxx_std_17)
target_compile_definitions(soui_host_focus_progress PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_compile_options(soui_host_focus_progress PRIVATE /EHsc /utf-8 /W4 /U_DEBUG)
set_property(TARGET soui_host_focus_progress PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
# Runtime execution is explicit: arguments bind a task-owned runtime, page,
# fresh output and expected node.dll SHA256. Missing runtime is not a passing test.
file(COPY resources DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
