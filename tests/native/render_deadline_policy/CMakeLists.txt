cmake_minimum_required(VERSION 3.24)
project(soui_render_deadline_policy LANGUAGES CXX)
enable_testing()
get_filename_component(SOUI_RENDER_POLICY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(soui_render_deadline_policy_tests ../render_deadline_policy_tests.cpp)
target_compile_features(soui_render_deadline_policy_tests PRIVATE cxx_std_17)
target_include_directories(soui_render_deadline_policy_tests PRIVATE
  "${SOUI_RENDER_POLICY_ROOT}/third_party/sources/miniblink")
if(MSVC)
  target_compile_options(soui_render_deadline_policy_tests PRIVATE /W4 /WX /EHsc)
  set_property(TARGET soui_render_deadline_policy_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_test(NAME kernel.native.legacy-paint-deadline COMMAND soui_render_deadline_policy_tests)
set_tests_properties(kernel.native.legacy-paint-deadline PROPERTIES TIMEOUT 10)
