cmake_minimum_required(VERSION 3.24)
project(SouiExternalConsumer LANGUAGES CXX)
find_package(SouiSdk CONFIG REQUIRED)
add_executable(soui_consumer main.cpp)
target_link_libraries(soui_consumer PRIVATE Soui::Sdk)
target_compile_options(soui_consumer PRIVATE /U_DEBUG)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-library-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:Soui::Sdk>\n")
set_property(TARGET soui_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(soui_devtools_consumer devtools.cpp)
target_link_libraries(soui_devtools_consumer PRIVATE Soui::Sdk)
target_compile_options(soui_devtools_consumer PRIVATE /utf-8 /U_DEBUG /EHsc)
set_property(TARGET soui_devtools_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
