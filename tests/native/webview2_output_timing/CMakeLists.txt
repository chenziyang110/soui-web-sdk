cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiWebView2OutputTiming LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(_timing_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_program(_timing_node node REQUIRED)
set(_timing_generated "${CMAKE_CURRENT_BINARY_DIR}/projection")
execute_process(COMMAND "${_timing_node}" "${CMAKE_CURRENT_LIST_DIR}/source-projection.cjs"
  "${_timing_root}/sdk/renderers/webview2/webview2_host.cpp" "${_timing_generated}" RESULT_VARIABLE _projection_status)
if(NOT _projection_status EQUAL 0)
  message(FATAL_ERROR "WebView2 serialization projection refused current source")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${_timing_root}/sdk/renderers/webview2/webview2_host.cpp" "${CMAKE_CURRENT_LIST_DIR}/source-projection.cjs")
add_executable(soui_webview2_output_timing_tests fixture.cpp
  "${_timing_root}/sdk/src/runtime/diagnostics/input_frame_trace.cpp")
target_include_directories(soui_webview2_output_timing_tests PRIVATE "${_timing_generated}" "${_timing_root}/sdk/include")
add_executable(soui_webview2_output_cpu_bench cpu-bench.cpp)
target_include_directories(soui_webview2_output_cpu_bench PRIVATE "${_timing_generated}" "${_timing_root}/third_party/include/json" "${_timing_root}/sdk/include")
foreach(_target soui_webview2_output_timing_tests soui_webview2_output_cpu_bench)
  target_compile_features(${_target} PRIVATE cxx_std_17)
  target_compile_definitions(${_target} PRIVATE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_compile_options(${_target} PRIVATE /W4 /utf-8 /EHsc)
  set_property(TARGET ${_target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endforeach()
add_test(NAME sdk.native.webview2-output-timing COMMAND soui_webview2_output_timing_tests)
set_tests_properties(sdk.native.webview2-output-timing PROPERTIES TIMEOUT 10)
add_test(NAME sdk.native.webview2-output-timing-reader COMMAND "${_timing_node}"
  "${CMAKE_CURRENT_LIST_DIR}/check-fixture.cjs" "$<TARGET_FILE:soui_webview2_output_timing_tests>" "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(sdk.native.webview2-output-timing-reader PROPERTIES TIMEOUT 15)
