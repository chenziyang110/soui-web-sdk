cmake_minimum_required(VERSION 3.24)
project(SouiTrayRecoveryConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Tray recovery consumer requires the supported Windows/MSVC SDK toolchain")
endif()
find_package(SouiSdk 0.6.0 EXACT CONFIG REQUIRED)
add_executable(SouiTrayRecoveryConsumer main.cpp)
target_link_libraries(SouiTrayRecoveryConsumer PRIVATE Soui::Sdk)
target_compile_features(SouiTrayRecoveryConsumer PRIVATE cxx_std_17)
target_compile_options(SouiTrayRecoveryConsumer PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiTrayRecoveryConsumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiTrayRecoveryConsumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
