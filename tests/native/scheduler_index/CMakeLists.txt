cmake_minimum_required(VERSION 3.24)
project(SouiSchedulerIndex LANGUAGES CXX)
enable_testing()
if(NOT DEFINED SOUI_DISTRIBUTION_ROOT)
  get_filename_component(SOUI_DISTRIBUTION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
endif()
add_executable(scheduler_index_contract fixture.cpp)
target_compile_features(scheduler_index_contract PRIVATE cxx_std_17)
target_include_directories(scheduler_index_contract PRIVATE "${SOUI_DISTRIBUTION_ROOT}/third_party/sources/miniblink")
add_test(NAME scheduler.index.contract COMMAND scheduler_index_contract)
