cmake_minimum_required(VERSION 3.24)
if(SOUI_BUILD_DESKTOP AND CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum system for the opt-in desktop backend")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/soui/version.h" SOUI_VERSION_DECLARATION REGEX "^#define SOUI_SDK_VERSION_STRING")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" SOUI_PROJECT_VERSION "${SOUI_VERSION_DECLARATION}")
project(SouiSdk VERSION ${SOUI_PROJECT_VERSION} LANGUAGES C CXX)
get_filename_component(SOUI_DISTRIBUTION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SOUI_DISTRIBUTION_ROOT "${SOUI_DISTRIBUTION_ROOT}" CACHE INTERNAL "SDK distribution root")
include("${SOUI_DISTRIBUTION_ROOT}/cmake/SouiPlatform.cmake")
soui_detect_platform()
option(SOUI_BUILD_PORTABLE "Build the portable Runtime without a desktop renderer" OFF)
option(SOUI_BUILD_DESKTOP "Build the unified system-WebView desktop SDK" OFF)
if(SOUI_BUILD_PORTABLE OR SOUI_BUILD_DESKTOP)
  if(SOUI_BUILD_WEBVIEW2 OR SOUI_BUILD_UIA)
    message(FATAL_ERROR "Portable Runtime cannot enable Windows WebView2 or UIA components")
  endif()
  soui_require_capability(portable_core)
  include(cmake/SouiPortableRuntime.cmake)
  if(SOUI_BUILD_DESKTOP)
    add_subdirectory(desktop)
  endif()
  return()
endif()
soui_require_capability(desktop_host)
if(SOUI_NATIVE_TOOLCHAIN STREQUAL "LLVM22MSVC142")
  if(NOT COMMAND soui_llvm22_verify_project)
    message(FATAL_ERROR "The explicit LLVM SDK profile requires the product toolchain")
  endif()
  soui_llvm22_verify_project()
endif()
if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "SouiSdk currently requires Windows and the MSVC v143 toolchain")
endif()
file(GLOB_RECURSE SOUI_NATIVE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
# The opt-in system-WebView entry owns platform-specific sources separately.
list(FILTER SOUI_NATIVE_SOURCES EXCLUDE REGEX "/src/desktop/")
add_library(soui_sdk STATIC ${SOUI_NATIVE_SOURCES})
add_library(Soui::Sdk ALIAS soui_sdk)
target_compile_features(soui_sdk PUBLIC cxx_std_17)
target_include_directories(soui_sdk PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<INSTALL_INTERFACE:include>")
target_compile_definitions(soui_sdk PUBLIC UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_compile_definitions(soui_sdk PRIVATE ENABLE_WKE=0 SOUI_SOURCE_KERNEL_HOST NDEBUG _CRT_SECURE_NO_WARNINGS)
target_compile_options(soui_sdk PRIVATE /utf-8 /W4 /U_DEBUG)
set_property(TARGET soui_sdk PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(soui_sdk PUBLIC user32 gdi32 imm32 shell32 ole32 oleaut32 uuid comctl32 advapi32 winmm bcrypt)
set_target_properties(soui_sdk PROPERTIES EXPORT_NAME Sdk)
# Header-only dependency for generated bindings and the local Agent endpoint.
add_library(soui_action_json INTERFACE)
add_library(Soui::ActionJson ALIAS soui_action_json)
target_compile_features(soui_action_json INTERFACE cxx_std_17)
target_include_directories(soui_action_json INTERFACE
  "$<BUILD_INTERFACE:${SOUI_DISTRIBUTION_ROOT}/third_party/include/json>"
  "$<INSTALL_INTERFACE:include/soui/action-json>")
set_target_properties(soui_action_json PROPERTIES EXPORT_NAME ActionJson)
target_link_libraries(soui_sdk PRIVATE Soui::ActionJson)
option(SOUI_BUILD_WEBVIEW2 "Build the optional modern WebView2 backend" OFF)
if(SOUI_BUILD_WEBVIEW2)
  add_subdirectory(renderers/webview2)
endif()
option(SOUI_BUILD_UIA "Build the optional finite Windows UIA provider" OFF)
if(SOUI_BUILD_UIA)
  add_subdirectory(accessibility/uia)
endif()

set(ZLIB_SOURCE "${SOUI_DISTRIBUTION_ROOT}/third_party/sources/zlib")
add_library(soui_zlib STATIC ${ZLIB_SOURCE}/adler32.c ${ZLIB_SOURCE}/compress.c
  ${ZLIB_SOURCE}/crc32.c ${ZLIB_SOURCE}/deflate.c ${ZLIB_SOURCE}/infback.c
  ${ZLIB_SOURCE}/inffast.c ${ZLIB_SOURCE}/inflate.c ${ZLIB_SOURCE}/inftrees.c
  ${ZLIB_SOURCE}/trees.c ${ZLIB_SOURCE}/uncompr.c ${ZLIB_SOURCE}/zutil.c)
target_include_directories(soui_zlib PUBLIC "$<BUILD_INTERFACE:${ZLIB_SOURCE}>")
target_compile_definitions(soui_zlib PRIVATE _CRT_SECURE_NO_WARNINGS)
set_property(TARGET soui_zlib PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)

include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/SouiSdkConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/SouiSdkConfig.cmake" INSTALL_DESTINATION lib/cmake/SouiSdk)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SouiSdkConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(TARGETS soui_sdk soui_action_json EXPORT SouiSdkTargets ARCHIVE DESTINATION lib/$<CONFIG>)
install(FILES "${SOUI_DISTRIBUTION_ROOT}/third_party/include/json/json.hpp" DESTINATION include/soui/action-json)
install(DIRECTORY include/ DESTINATION include)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOUI_DISTRIBUTION_ROOT}/THIRD_PARTY_NOTICES.md")
file(READ "${SOUI_DISTRIBUTION_ROOT}/THIRD_PARTY_NOTICES.md" _soui_install_notices)
string(REPLACE "](third_party/licenses/" "](licenses/"
  _soui_install_notices "${_soui_install_notices}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/THIRD_PARTY_NOTICES.install.md" "${_soui_install_notices}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/THIRD_PARTY_NOTICES.install.md"
  DESTINATION share/SouiSdk RENAME THIRD_PARTY_NOTICES.md)
install(DIRECTORY "${SOUI_DISTRIBUTION_ROOT}/third_party/licenses/" DESTINATION share/SouiSdk/licenses)
install(EXPORT SouiSdkTargets NAMESPACE Soui:: DESTINATION lib/cmake/SouiSdk)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SouiSdkConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/SouiSdkConfigVersion.cmake" DESTINATION lib/cmake/SouiSdk)
