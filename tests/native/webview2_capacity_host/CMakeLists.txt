# Actual WebView2 browser + normal installed/source SDK; no copied implementation.
option(SOUI_BUILD_WEBVIEW2_CAPACITY_HOST_TESTS "Run actual WebView2 capacity reserve host" OFF)
if(NOT SOUI_BUILD_WEBVIEW2_CAPACITY_HOST_TESTS)
  return()
endif()
if(NOT TARGET Soui::Sdk OR NOT TARGET Soui::WebView2 OR NOT TARGET Soui::ActionJson)
  message(FATAL_ERROR "Actual WebView2 capacity tests require normal SDK, WebView2 and ActionJson")
endif()
add_executable(soui_webview2_capacity_host_tests capacity_host.cpp)
target_compile_features(soui_webview2_capacity_host_tests PRIVATE cxx_std_17)
target_compile_options(soui_webview2_capacity_host_tests PRIVATE /W4 /EHsc /utf-8 /U_DEBUG)
target_compile_definitions(soui_webview2_capacity_host_tests PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
set_property(TARGET soui_webview2_capacity_host_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(soui_webview2_capacity_host_tests PRIVATE Soui::WebView2 Soui::ActionJson user32 ole32)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/resources" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
set(SOUI_WEBVIEW2_CAPACITY_WEB_DIST "${SOUI_DISTRIBUTION_ROOT}/sdk/web/dist" CACHE PATH "Actual normal SDK Web build output")
if(NOT IS_ABSOLUTE "${SOUI_WEBVIEW2_CAPACITY_WEB_DIST}" OR NOT EXISTS "${SOUI_WEBVIEW2_CAPACITY_WEB_DIST}/index.js")
  message(FATAL_ERROR "Build the real SDK Web module normally before configuring the actual Host test")
endif()
file(COPY "${SOUI_WEBVIEW2_CAPACITY_WEB_DIST}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/resources/soui-web-sdk")
set(SOUI_WEBVIEW2_CAPACITY_FIXED_RUNTIME "" CACHE PATH "Optional actual fixed WebView2 runtime; empty uses installed runtime")
set(capacity_runtime_args)
if(SOUI_WEBVIEW2_CAPACITY_FIXED_RUNTIME)
  if(NOT IS_ABSOLUTE "${SOUI_WEBVIEW2_CAPACITY_FIXED_RUNTIME}" OR NOT IS_DIRECTORY "${SOUI_WEBVIEW2_CAPACITY_FIXED_RUNTIME}")
    message(FATAL_ERROR "Fixed WebView2 runtime must be an absolute actual directory")
  endif()
  list(APPEND capacity_runtime_args --runtime "${SOUI_WEBVIEW2_CAPACITY_FIXED_RUNTIME}")
endif()
add_test(NAME sdk.native.webview2-host-capacity.reserve-recovery COMMAND soui_webview2_capacity_host_tests
  --profile "${CMAKE_CURRENT_BINARY_DIR}/profile" --resources "${CMAKE_CURRENT_BINARY_DIR}/resources"
  --output "${CMAKE_CURRENT_BINARY_DIR}/result.json" ${capacity_runtime_args})
# The existing actual WebView2 preparation test also uses 60 seconds. Every wire
# request still uses the unchanged 10-second transport budget; no long renewal.
set_tests_properties(sdk.native.webview2-host-capacity.reserve-recovery PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
