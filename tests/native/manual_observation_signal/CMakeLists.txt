cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiManualObservationSignal LANGUAGES CXX)
  enable_testing()
endif()
if(NOT WIN32)
  message(FATAL_ERROR "Manual observation signal requires Win32")
endif()
if(NOT DEFINED SOUI_DISTRIBUTION_ROOT)
  get_filename_component(SOUI_DISTRIBUTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
endif()
add_executable(soui_manual_observation_signal_tests
  "${SOUI_DISTRIBUTION_ROOT}/tests/benchmark/manual-observation-signal.cpp")
target_compile_features(soui_manual_observation_signal_tests PRIVATE cxx_std_17)
target_include_directories(soui_manual_observation_signal_tests PRIVATE
  "${SOUI_DISTRIBUTION_ROOT}/benchmarks/protocol")
target_link_libraries(soui_manual_observation_signal_tests PRIVATE user32)
if(MSVC)
  target_compile_options(soui_manual_observation_signal_tests PRIVATE /EHsc /utf-8 /W4 /U_DEBUG)
  set_property(TARGET soui_manual_observation_signal_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
add_test(NAME benchmark.manual-observation-signal COMMAND soui_manual_observation_signal_tests)
set_tests_properties(benchmark.manual-observation-signal PROPERTIES TIMEOUT 10)
