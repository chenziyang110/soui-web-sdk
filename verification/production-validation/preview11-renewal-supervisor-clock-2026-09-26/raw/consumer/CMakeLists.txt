cmake_minimum_required(VERSION 3.24)
project(InstalledRenewalProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "InstalledRenewalProbe requires the supported Windows/MSVC toolchain")
endif()
find_package(SouiSdk 0.3.0 EXACT CONFIG REQUIRED)
add_executable(InstalledRenewalProbe main.cpp)
target_compile_features(InstalledRenewalProbe PRIVATE cxx_std_17)
target_compile_definitions(InstalledRenewalProbe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(InstalledRenewalProbe PRIVATE /W4 /utf-8 /U_DEBUG)
target_link_libraries(InstalledRenewalProbe PRIVATE Soui::Sdk psapi ole32)
set_property(TARGET InstalledRenewalProbe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(InstalledRenewalProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")

add_executable(SupervisorClockTests supervisor_clock_tests.cpp)
target_compile_features(SupervisorClockTests PRIVATE cxx_std_17)
target_compile_options(SupervisorClockTests PRIVATE /W4 /utf-8)
set_property(TARGET SupervisorClockTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
