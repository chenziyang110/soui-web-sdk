cmake_minimum_required(VERSION 3.24)
project(TrayNegativeControl LANGUAGES CXX)
add_executable(tray_negative_control tray_service.cpp tray_service_tests.cpp)
target_include_directories(tray_negative_control PRIVATE "${SDK_ROOT}/sdk/include" "${SDK_ROOT}/sdk/src" "${SDK_ROOT}/sdk/src/runtime/services/desktop")
target_compile_features(tray_negative_control PRIVATE cxx_std_17)
target_compile_options(tray_negative_control PRIVATE /W4 /utf-8 /U_DEBUG)
target_compile_definitions(tray_negative_control PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
set_property(TARGET tray_negative_control PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(tray_negative_control PRIVATE user32 shell32)
