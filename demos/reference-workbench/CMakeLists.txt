cmake_minimum_required(VERSION 3.24)
project(SouiReferenceWorkbench VERSION 0.5.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "The reference application requires Windows and MSVC")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The reference application supports x86 and x64 targets")
endif()
set(SOUI_REFERENCE_DISTRIBUTION_ROOT "" CACHE PATH "Current complete source delivery for Web tools, notices and runtime inputs")
set(SOUI_REFERENCE_MINIBLINK "" CACHE FILEPATH "Explicit current verified software kernel for compat deployment")
option(SOUI_REFERENCE_MODERN "Build optional Modern/WebView2 reference" OFF)
option(SOUI_REFERENCE_COMPAT "Build React18 and Vue3 Miniblink references" ON)
if(NOT EXISTS "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/tools/sdk/actiongen.cjs")
  message(FATAL_ERROR "Pass the current complete source delivery root")
endif()
if(NOT TARGET Soui::Sdk)
  if(SOUI_REFERENCE_MODERN)
    find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS ActionJson WebView2)
  else()
    find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS ActionJson)
  endif()
endif()
find_program(SOUI_REFERENCE_NODE node.exe REQUIRED)
add_custom_target(reference_actions_check ALL
  COMMAND "${SOUI_REFERENCE_NODE}" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/tools/sdk/actiongen.cjs"
    --source "${CMAKE_CURRENT_SOURCE_DIR}/contracts/actions.json" --root "${CMAKE_CURRENT_SOURCE_DIR}" --check VERBATIM)
add_library(reference_business STATIC app/reference-business.cpp)
target_link_libraries(reference_business PUBLIC Soui::Sdk Soui::ActionJson)
target_compile_features(reference_business PUBLIC cxx_std_17)
target_compile_options(reference_business PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
set_property(TARGET reference_business PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_include_directories(reference_business PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/app")
add_dependencies(reference_business reference_actions_check)
add_library(reference_agent STATIC app/reference-agent.cpp)
target_link_libraries(reference_agent PUBLIC reference_business PRIVATE advapi32)
target_compile_features(reference_agent PUBLIC cxx_std_17)
target_compile_options(reference_agent PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
set_property(TARGET reference_agent PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(reference_arch x64)
else()
  set(reference_arch x86)
endif()
file(GLOB_RECURSE reference_crt "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/third_party/prebuilt/${reference_arch}/runtime/*.dll")
if(NOT reference_crt)
  message(FATAL_ERROR "Current deployment CRT inputs are required; no old output fallback")
endif()
function(reference_app name web)
  add_executable(${name} WIN32 app/main.cpp app/application.manifest)
  target_link_libraries(${name} PRIVATE reference_agent shell32 ole32)
  target_compile_options(${name} PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_options(${name} PRIVATE /SUBSYSTEM:WINDOWS,6.01)
  set_target_properties(${name} PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/${name}")
  if(NOT EXISTS "${web}/index.html" OR NOT EXISTS "${web}/reference-fixture.json")
    message(FATAL_ERROR "Build current exact Web view before native configure: ${web}")
  endif()
  add_custom_target(${name}_resources ALL
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${reference_crt} "$<TARGET_FILE_DIR:${name}>"
    COMMAND "${SOUI_REFERENCE_NODE}" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/tools/internal/deploy-resources.cjs"
      "${web}" "$<TARGET_FILE_DIR:${name}>" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}"
    DEPENDS ${name} VERBATIM)
endfunction()
if(SOUI_REFERENCE_COMPAT)
  if(NOT EXISTS "${SOUI_REFERENCE_MINIBLINK}")
    message(FATAL_ERROR "Explicit verified current kernel path is required")
  endif()
  foreach(view IN ITEMS react vue)
    if(view STREQUAL react)
      set(name SouiReferenceCompatReact)
    else()
      set(name SouiReferenceCompatVue)
    endif()
    reference_app(${name} "${CMAKE_CURRENT_SOURCE_DIR}/web-compat/dist-${view}")
    add_custom_command(TARGET ${name} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${SOUI_REFERENCE_MINIBLINK}" "$<TARGET_FILE_DIR:${name}>/node.dll" VERBATIM)
  endforeach()
endif()
if(SOUI_REFERENCE_MODERN)
  if(NOT TARGET Soui::WebView2)
    message(FATAL_ERROR "Modern reference requires the explicit optional current WebView2 SDK target")
  endif()
  reference_app(SouiReferenceModern "${CMAKE_CURRENT_SOURCE_DIR}/web-modern/dist")
  target_compile_definitions(SouiReferenceModern PRIVATE SOUI_REFERENCE_MODERN)
  target_link_libraries(SouiReferenceModern PRIVATE Soui::WebView2)
endif()
