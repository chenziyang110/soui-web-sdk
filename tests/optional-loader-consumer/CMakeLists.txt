cmake_minimum_required(VERSION 3.24)
project(SouiOptionalLoaderConsumer LANGUAGES CXX)
find_package(SouiSdk 0.6.1 EXACT CONFIG REQUIRED COMPONENTS AutoHost)
add_executable(SouiOptionalLoaderConsumer main.cpp)
target_link_libraries(SouiOptionalLoaderConsumer PRIVATE Soui::AutoHost)
target_compile_features(SouiOptionalLoaderConsumer PRIVATE cxx_std_17)
target_compile_options(SouiOptionalLoaderConsumer PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiOptionalLoaderConsumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
soui_deploy_webview2_loader(SouiOptionalLoaderConsumer)
