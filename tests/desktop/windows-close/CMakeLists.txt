cmake_minimum_required(VERSION 3.24)
project(SouiWindowsDomCloseRegression LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This regression requires real Windows/WebView2")
endif()
find_package(SouiDesktop CONFIG REQUIRED)
add_executable(soui_windows_dom_close main.cpp)
target_link_libraries(soui_windows_dom_close PRIVATE Soui::Desktop)
target_compile_features(soui_windows_dom_close PRIVATE cxx_std_17)
target_compile_options(soui_windows_dom_close PRIVATE /utf-8 /W4)
set_property(TARGET soui_windows_dom_close PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_include_directories(soui_windows_dom_close PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../third_party/include/json")
add_custom_command(TARGET soui_windows_dom_close POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:soui_windows_dom_close>/web"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/desktop/frontend/desktop.mjs" "$<TARGET_FILE_DIR:soui_windows_dom_close>/web/desktop.mjs")
enable_testing()
option(SOUI_DESKTOP_GUI_TESTS "Register tests requiring a real interactive Windows desktop" OFF)
if(SOUI_DESKTOP_GUI_TESTS)
  add_test(NAME sdk.desktop.windows-dom-close COMMAND soui_windows_dom_close)
  set_tests_properties(sdk.desktop.windows-dom-close PROPERTIES TIMEOUT 40 LABELS "desktop;native-gui")
endif()
