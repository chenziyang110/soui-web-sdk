cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SOUI_DELIVERY_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" SOUI_PROJECT_VERSION "${SOUI_DELIVERY_VERSION}")
project(SouiSdkDistribution VERSION ${SOUI_PROJECT_VERSION} LANGUAGES C CXX RC)
option(SOUI_BUILD_TEMPLATE "Build the desktop starter template" ON)
option(SOUI_BUILD_DEMOS "Build the workbench demonstration" ON)
option(SOUI_BUILD_TESTS "Build native SDK tests" ON)
include(cmake/SouiVendorTests.cmake)
add_subdirectory(sdk)
add_subdirectory(tools/runtime-doctor)
include(cmake/SouiApplication.cmake)
option(SOUI_BUILD_INPUT_UI_BENCHMARKS "Build the explicit public SDK input/UI measurement host" OFF)
if(SOUI_BUILD_INPUT_UI_BENCHMARKS)
  add_subdirectory(benchmarks/input-ui-host)
endif()
if(SOUI_BUILD_TEMPLATE)
  add_subdirectory(templates/desktop-app)
  if(SOUI_BUILD_WEBVIEW2)
    add_subdirectory(templates/modern-app)
  endif()
endif()
if(SOUI_BUILD_DEMOS)
  add_subdirectory(demos/workbench)
endif()
if(SOUI_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/native)
endif()
