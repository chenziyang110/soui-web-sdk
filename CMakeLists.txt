cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SOUI_DELIVERY_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" SOUI_PROJECT_VERSION "${SOUI_DELIVERY_VERSION}")
project(SouiSdkDistribution VERSION ${SOUI_PROJECT_VERSION} LANGUAGES C CXX)
option(SOUI_BUILD_PORTABLE "Build the portable Runtime without a desktop renderer" OFF)
if(SOUI_BUILD_PORTABLE)
  set(_soui_desktop_default OFF)
else()
  set(_soui_desktop_default ON)
endif()
option(SOUI_BUILD_TEMPLATE "Build the desktop starter template" ${_soui_desktop_default})
option(SOUI_BUILD_DEMOS "Build the workbench demonstration" ${_soui_desktop_default})
option(SOUI_BUILD_TESTS "Build native SDK tests" ON)
if(SOUI_BUILD_PORTABLE)
  if(SOUI_BUILD_TEMPLATE OR SOUI_BUILD_DEMOS OR SOUI_BUILD_WEBVIEW2 OR SOUI_BUILD_UIA OR SOUI_BUILD_VENDOR_TESTS)
    message(FATAL_ERROR "Portable Runtime does not provide desktop templates, demos, WebView2, UIA or vendor tests. Use a separate build directory with these options OFF.")
  endif()
  add_subdirectory(sdk)
  if(SOUI_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests/portable)
  endif()
  return()
endif()
include(cmake/SouiVendorTests.cmake)
add_subdirectory(sdk)
enable_language(RC)
add_subdirectory(tools/runtime-doctor)
include(cmake/SouiApplication.cmake)
if(SOUI_BUILD_TEMPLATE)
  add_subdirectory(templates/desktop-app)
  if(SOUI_BUILD_WEBVIEW2)
    add_subdirectory(templates/modern-app)
  endif()
endif()
if(SOUI_BUILD_DEMOS)
  add_subdirectory(demos/workbench)
endif()
if(SOUI_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/native)
endif()
