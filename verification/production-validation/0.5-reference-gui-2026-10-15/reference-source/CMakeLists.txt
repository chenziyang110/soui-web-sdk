project(SouiReferenceWorkbench VERSION 0.5.0 LANGUAGES CXX)

set_property(CACHE SOUI_REFERENCE_NATIVE_TOOLCHAIN PROPERTY STRINGS MSVC143 LLVM22MSVC142)
set(SOUI_REFERENCE_LLVM_SDK_REQUEST "" CACHE FILEPATH "Current naturally closed standard LLVM SDK Build request")
set(SOUI_REFERENCE_SDK_CONFIGURATION "Release" CACHE STRING "Installed SDK configuration verified for the LLVM reference")
if(SOUI_REFERENCE_NATIVE_TOOLCHAIN STREQUAL "LLVM22MSVC142")
  if(CMAKE_VERSION VERSION_LESS 3.29 OR NOT WIN32 OR NOT MSVC OR
      NOT CMAKE_GENERATOR STREQUAL "Ninja Multi-Config" OR NOT SOUI_NATIVE_TOOLCHAIN STREQUAL "LLVM22MSVC142" OR
      NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 22.1.3 OR
      NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC" OR NOT CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC" OR
      NOT CMAKE_CXX_SIMULATE_VERSION VERSION_EQUAL 19.29 OR NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.19041.0")
    message(FATAL_ERROR "LLVM reference requires the exact verified LLVM22MSVC142 profile")
  endif()
  if(NOT SOUI_REFERENCE_SDK_CONFIGURATION MATCHES "^(Debug|Release)$" OR NOT IS_ABSOLUTE "${SOUI_REFERENCE_LLVM_SDK_REQUEST}")
    message(FATAL_ERROR "LLVM reference requires an explicit current SDK request and exact Debug or Release configuration")
  endif()
elseif(NOT SOUI_REFERENCE_NATIVE_TOOLCHAIN STREQUAL "MSVC143" OR SOUI_REFERENCE_LLVM_SDK_REQUEST OR
    NOT WIN32 OR NOT MSVC OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "The default reference application requires Windows and MSVC; LLVM requires explicit opt-in")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The reference application supports x86 and x64 targets")
endif()

    find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS ActionJson)
  endif()

if(SOUI_REFERENCE_NATIVE_TOOLCHAIN STREQUAL "LLVM22MSVC142")
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(reference_profile_arch x64)
    set(reference_profile_target x86_64-pc-windows-msvc)
  else()
    set(reference_profile_arch x86)
    set(reference_profile_target i686-pc-windows-msvc)
  endif()
  if(NOT SOUI_LLVM_ARCHITECTURE STREQUAL reference_profile_arch OR NOT CMAKE_CXX_COMPILER_TARGET STREQUAL reference_profile_target)
    message(FATAL_ERROR "LLVM reference target and actual pointer width differ")
  endif()
  get_filename_component(reference_toolchain "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/cmake/toolchains/SouiLLVM22.cmake" ABSOLUTE)
  if(NOT CMAKE_TOOLCHAIN_FILE STREQUAL reference_toolchain)
    message(FATAL_ERROR "LLVM reference must use the fixed current distribution toolchain")
  endif()
  get_target_property(reference_imported Soui::Sdk IMPORTED)
  if(NOT reference_imported)
    message(FATAL_ERROR "LLVM reference must consume an installed SDK, without SDK source TUs")
  endif()
  include("${SOUI_REFERENCE_DISTRIBUTION_ROOT}/cmake/SouiReferenceSdkConfiguration.cmake")
  soui_reference_sdk_configuration(Soui::Sdk "${SOUI_REFERENCE_SDK_CONFIGURATION}")
  if(SOUI_REFERENCE_MODERN)
    soui_reference_sdk_configuration(Soui::WebView2 "${SOUI_REFERENCE_SDK_CONFIGURATION}")
  endif()
  string(TOUPPER "${SOUI_REFERENCE_SDK_CONFIGURATION}" reference_sdk_config)
  get_target_property(reference_sdk_archive Soui::Sdk "IMPORTED_LOCATION_${reference_sdk_config}")
  get_target_property(reference_sdk_includes Soui::Sdk INTERFACE_INCLUDE_DIRECTORIES)
  list(LENGTH reference_sdk_includes reference_include_count)
  if(NOT reference_include_count EQUAL 1 OR NOT IS_ABSOLUTE "${reference_sdk_archive}")
    message(FATAL_ERROR "LLVM reference requires the exact installed archive and one public include root")
  endif()
  execute_process(COMMAND "${SOUI_REFERENCE_NODE}" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/tools/internal/llvm-reference-sdk.cjs"
      verify "${SOUI_REFERENCE_DISTRIBUTION_ROOT}" "${reference_profile_arch}" "${SOUI_REFERENCE_SDK_CONFIGURATION}"
      "${SOUI_REFERENCE_LLVM_SDK_REQUEST}" "${reference_sdk_archive}" "${reference_sdk_includes}"
    RESULT_VARIABLE reference_profile_result OUTPUT_VARIABLE reference_profile_proof ERROR_VARIABLE reference_profile_error TIMEOUT 120)
  if(NOT "${reference_profile_result}" STREQUAL "0")
    message(FATAL_ERROR "Current LLVM SDK build/install proof rejected: ${reference_profile_error}")
  endif()
  string(JSON reference_input_identity ERROR_VARIABLE reference_json_error GET "${reference_profile_proof}" inputIdentity)
  string(JSON reference_bridge_identity ERROR_VARIABLE reference_bridge_error GET "${reference_profile_proof}" bridgeIdentity)
  string(JSON reference_toolchain_sha ERROR_VARIABLE reference_toolchain_error GET "${reference_profile_proof}" toolchain sha256)
  file(SHA256 "${reference_toolchain}" reference_actual_toolchain_sha)
  if(reference_json_error OR reference_bridge_error OR reference_toolchain_error OR
      NOT reference_input_identity STREQUAL SOUI_LLVM_INPUT_IDENTITY OR NOT reference_bridge_identity STREQUAL SOUI_LLVM_BRIDGE_IDENTITY OR
      NOT reference_actual_toolchain_sha STREQUAL reference_toolchain_sha)
    message(FATAL_ERROR "Reference compiler request differs from the current verified SDK profile")
  endif()
endif()

function(reference_app name web)
  add_executable(${name} WIN32 app/main.cpp app/application.manifest)
  target_link_libraries(${name} PRIVATE reference_agent shell32 ole32)
  target_compile_options(${name} PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_options(${name} PRIVATE /SUBSYSTEM:WINDOWS,6.01)
  set_target_properties(${name} PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/${name}")
  if(NOT EXISTS "${web}/index.html" OR NOT EXISTS "${web}/reference-fixture.json")
    message(FATAL_ERROR "Build current exact Web view before native configure: ${web}")
  endif()
  add_custom_target(${name}_resources ALL
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${reference_crt} "$<TARGET_FILE_DIR:${name}>"
    COMMAND "${SOUI_REFERENCE_NODE}" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}/tools/internal/deploy-resources.cjs"
      "${web}" "$<TARGET_FILE_DIR:${name}>" "${SOUI_REFERENCE_DISTRIBUTION_ROOT}"
    DEPENDS ${name} VERBATIM)
endfunction()
if(SOUI_REFERENCE_COMPAT)
  if(NOT EXISTS "${SOUI_REFERENCE_MINIBLINK}")
    message(FATAL_ERROR "Explicit verified current kernel path is required")
  endif()
  foreach(view IN ITEMS react vue)
    if(view STREQUAL react)
      set(name SouiReferenceCompatReact)
    else()
      set(name SouiReferenceCompatVue)
    endif()
    reference_app(${name} "${CMAKE_CURRENT_SOURCE_DIR}/web-compat/dist-${view}")
    add_custom_command(TARGET ${name} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${SOUI_REFERENCE_MINIBLINK}" "$<TARGET_FILE_DIR:${name}>/node.dll" VERBATIM)
  endforeach()
endif()
