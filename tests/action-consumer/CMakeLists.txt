cmake_minimum_required(VERSION 3.24)
project(SouiGeneratedActionConsumer LANGUAGES CXX)
find_package(SouiSdk CONFIG REQUIRED COMPONENTS ActionJson)
add_executable(soui_action_consumer main.cpp)
target_link_libraries(soui_action_consumer PRIVATE Soui::Sdk Soui::ActionJson)
target_compile_options(soui_action_consumer PRIVATE /utf-8 /W4 /U_DEBUG)
set_property(TARGET soui_action_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
enable_testing()
add_test(NAME sdk.consumer.generated-actions COMMAND soui_action_consumer)
set_tests_properties(sdk.consumer.generated-actions PROPERTIES TIMEOUT 10)
