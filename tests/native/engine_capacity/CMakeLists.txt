# Kept separate from the generic Engine group: no missing-kernel pass.
option(SOUI_BUILD_ENGINE_CAPACITY_HOST_TESTS "Build actual mounted Host capacity tests (explicit kernel)" OFF)
if(SOUI_BUILD_ENGINE_CAPACITY_HOST_TESTS)
  if(NOT TARGET Soui::ActionJson)
    message(FATAL_ERROR "actual Host capacity tests require normal ActionJson")
  endif()
  if(NOT IS_ABSOLUTE "${SOUI_ENGINE_CAPACITY_KERNEL}" OR NOT EXISTS "${SOUI_ENGINE_CAPACITY_KERNEL}")
    message(FATAL_ERROR "actual Host capacity tests require an absolute real kernel")
  endif()
  if(NOT SOUI_ENGINE_CAPACITY_KERNEL_SHA256 MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "actual Host capacity tests require the exact SHA256")
  endif()
  string(LENGTH "${SOUI_ENGINE_CAPACITY_KERNEL_SHA256}" engine_kernel_hash_length)
  file(SHA256 "${SOUI_ENGINE_CAPACITY_KERNEL}" engine_kernel_actual_hash)
  if(NOT engine_kernel_hash_length EQUAL 64 OR NOT engine_kernel_actual_hash STREQUAL SOUI_ENGINE_CAPACITY_KERNEL_SHA256)
    message(FATAL_ERROR "actual Host capacity kernel pin changed")
  endif()
  if(NOT IS_ABSOLUTE "${SOUI_ENGINE_CAPACITY_RESOURCES}" OR NOT EXISTS "${SOUI_ENGINE_CAPACITY_RESOURCES}/index.html")
    message(FATAL_ERROR "Root must prepare a fresh output-owned resource copy")
  endif()
  file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/resources/index.html" engine_resource_expected_hash)
  file(SHA256 "${SOUI_ENGINE_CAPACITY_RESOURCES}/index.html" engine_resource_actual_hash)
  if(NOT engine_resource_expected_hash STREQUAL engine_resource_actual_hash)
    message(FATAL_ERROR "actual Host capacity resource body changed")
  endif()
  add_executable(soui_engine_host_capacity_tests host_capacity_fixture.cpp)
  target_compile_features(soui_engine_host_capacity_tests PRIVATE cxx_std_17)
  target_include_directories(soui_engine_host_capacity_tests PRIVATE "${SOUI_DISTRIBUTION_ROOT}/sdk/src/host")
  target_compile_options(soui_engine_host_capacity_tests PRIVATE /W4 /EHsc /utf-8 /U_DEBUG)
  set_property(TARGET soui_engine_host_capacity_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_link_libraries(soui_engine_host_capacity_tests PRIVATE Soui::Sdk Soui::ActionJson)
  foreach(case IN ITEMS reply-full-zero-business cleanup-full-retains-owner renderer-model-abi renderer-cache-abi)
    add_test(NAME sdk.native.engine-host-capacity.${case} COMMAND soui_engine_host_capacity_tests
      --kernel "${SOUI_ENGINE_CAPACITY_KERNEL}" --kernel-sha "${SOUI_ENGINE_CAPACITY_KERNEL_SHA256}"
      --resources "${SOUI_ENGINE_CAPACITY_RESOURCES}" --output "${CMAKE_CURRENT_BINARY_DIR}/${case}.json" --case ${case})
    set_tests_properties(sdk.native.engine-host-capacity.${case} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  endforeach()
endif()
