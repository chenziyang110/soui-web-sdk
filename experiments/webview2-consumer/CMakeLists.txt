cmake_minimum_required(VERSION 3.24)
project(SouiWebView2Consumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The experimental WebView2 consumer requires Windows/MSVC")
endif()
find_package(SouiSdk 0.3.0 EXACT CONFIG REQUIRED)
set(SOUI_WEBVIEW2_SDK_ROOT "" CACHE PATH "Externally restored Microsoft.Web.WebView2 package")
set(SOUI_JSON_INCLUDE_ROOT "" CACHE PATH "Directory containing the SDK's json.hpp")
set(SOUI_WEBVIEW2_WEB_ROOT "" CACHE PATH "Built experimental Web resource directory")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(webview2_arch x64)
else()
  set(webview2_arch x86)
endif()
find_path(WEBVIEW2_INCLUDE WebView2.h PATHS "${SOUI_WEBVIEW2_SDK_ROOT}/build/native/include" NO_DEFAULT_PATH REQUIRED)
find_library(WEBVIEW2_LOADER WebView2LoaderStatic PATHS "${SOUI_WEBVIEW2_SDK_ROOT}/build/native/${webview2_arch}" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${SOUI_JSON_INCLUDE_ROOT}/json.hpp")
  message(FATAL_ERROR "Set SOUI_JSON_INCLUDE_ROOT to the external JSON header directory")
endif()
if(NOT EXISTS "${SOUI_WEBVIEW2_WEB_ROOT}/index.html")
  message(FATAL_ERROR "Build the experimental page and set SOUI_WEBVIEW2_WEB_ROOT")
endif()
add_executable(SouiWebView2Consumer main.cpp webview2_adapter.cpp)
target_compile_features(SouiWebView2Consumer PRIVATE cxx_std_17)
target_compile_options(SouiWebView2Consumer PRIVATE /utf-8 /W4 /U_DEBUG)
target_include_directories(SouiWebView2Consumer PRIVATE "${WEBVIEW2_INCLUDE}" "${SOUI_JSON_INCLUDE_ROOT}")
target_link_libraries(SouiWebView2Consumer PRIVATE Soui::Sdk "${WEBVIEW2_LOADER}" shlwapi version bcrypt)
set_property(TARGET SouiWebView2Consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiWebView2Consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
add_custom_command(TARGET SouiWebView2Consumer POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/native-preload.js" "$<TARGET_FILE_DIR:SouiWebView2Consumer>/native-preload.js"
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${SOUI_WEBVIEW2_WEB_ROOT}" "$<TARGET_FILE_DIR:SouiWebView2Consumer>/resources"
  VERBATIM)
