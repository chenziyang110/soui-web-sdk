cmake_minimum_required(VERSION 3.24)
project(SouiProcessTreeEnumerationContracts LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "These contracts use Windows types; no GUI or process-tree sampling")
endif()
add_executable(ProcessTreeEnumerationTests fixture.cpp)
target_compile_features(ProcessTreeEnumerationTests PRIVATE cxx_std_17)
target_link_libraries(ProcessTreeEnumerationTests PRIVATE psapi)
if(MSVC)
  target_compile_options(ProcessTreeEnumerationTests PRIVATE /W4 /utf-8)
  set_property(TARGET ProcessTreeEnumerationTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  # clang-cl needs the forwarding form; a bare -f flag can be ignored.
  target_compile_options(ProcessTreeEnumerationTests PRIVATE /clang:-fno-elide-constructors)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  # Catches a close status incorrectly left for the destructor after return.
  target_compile_options(ProcessTreeEnumerationTests PRIVATE -fno-elide-constructors)
endif()
enable_testing()
add_test(NAME benchmark.process-tree-enumeration COMMAND ProcessTreeEnumerationTests)
set_tests_properties(benchmark.process-tree-enumeration PROPERTIES TIMEOUT 10)
