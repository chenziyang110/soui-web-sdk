cmake_minimum_required(VERSION 3.24)
project(SouiBenchmark LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The SOUI benchmark host requires Windows and MSVC")
endif()
find_package(SouiSdk 0.3.0 EXACT CONFIG REQUIRED)
add_executable(SouiBenchmark main.cpp)
target_link_libraries(SouiBenchmark PRIVATE Soui::Sdk)
target_compile_features(SouiBenchmark PRIVATE cxx_std_17)
target_compile_options(SouiBenchmark PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiBenchmark PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiBenchmark PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
