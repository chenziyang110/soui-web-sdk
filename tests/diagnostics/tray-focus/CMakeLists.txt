cmake_minimum_required(VERSION 3.24)
project(SouiTrayFocusDiagnostic LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Tray focus diagnostic requires Windows and the supported MSVC toolchain")
endif()
add_executable(SouiTrayFocusDiagnostic main.cpp)
target_compile_features(SouiTrayFocusDiagnostic PRIVATE cxx_std_17)
target_compile_definitions(SouiTrayFocusDiagnostic PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 NOMINMAX)
target_compile_options(SouiTrayFocusDiagnostic PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(SouiTrayFocusDiagnostic PRIVATE user32 shell32 advapi32 ole32 version)
set_property(TARGET SouiTrayFocusDiagnostic PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(SouiTrayFocusDiagnostic PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
