cmake_minimum_required(VERSION 3.29)
project(InstalledNativeUiaDesktop LANGUAGES CXX)
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED COMPONENTS Uia)
if(NOT TARGET Soui::Uia)
  message(FATAL_ERROR "Installed optional UIA component required")
endif()
get_target_property(imported Soui::Uia IMPORTED)
if(NOT imported)
  message(FATAL_ERROR "An installed SDK is required")
endif()
set(REFERENCE_DISTRIBUTION "" CACHE PATH "Frozen SDK source headers")
add_executable(NativeUiaHost host.cpp provider-focus-trace.cpp)
target_link_libraries(NativeUiaHost PRIVATE Soui::Uia Soui::ActionJson user32 gdi32 ole32 oleaut32)
add_executable(NativeUiaClient client.cpp)
target_include_directories(NativeUiaClient PRIVATE
  "${REFERENCE_DISTRIBUTION}/demos/reference-workbench/tests"
  "${REFERENCE_DISTRIBUTION}/sdk/include" "${REFERENCE_DISTRIBUTION}/third_party/include/json")
target_link_libraries(NativeUiaClient PRIVATE user32 ole32 oleaut32)
add_executable(NativeUiaOriginalRegression regression.cpp)
add_executable(NativeUiaRegression regression.cpp provider-candidate.cpp)
foreach(target NativeUiaOriginalRegression NativeUiaRegression)
  target_link_libraries(${target} PRIVATE Soui::Uia Soui::ActionJson user32 gdi32 ole32 oleaut32)
endforeach()
foreach(target NativeUiaHost NativeUiaClient NativeUiaOriginalRegression NativeUiaRegression)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_compile_options(${target} PRIVATE /EHsc /utf-8 /W4 /U_DEBUG)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endforeach()
