cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum unified Desktop macOS target")
endif()
project(SouiCiInstalledConsumer LANGUAGES CXX)

option(CI_RUN_RUNTIME_TEST "Register only the native no-GUI Runtime consumer" OFF)
if(NOT DEFINED SDK_PREFIX OR NOT IS_ABSOLUTE "${SDK_PREFIX}" OR
   NOT IS_DIRECTORY "${SDK_PREFIX}")
  message(FATAL_ERROR "Provide an existing absolute SDK_PREFIX")
endif()
file(REAL_PATH "${SDK_PREFIX}" _prefix)
file(REAL_PATH "${_prefix}/include" _installed_include)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)

# Resolve Runtime first so Desktop's find_dependency cannot select a source-tree
# package or another installation. System platform libraries remain dependencies.
find_package(SouiRuntime 0.6.0 EXACT CONFIG REQUIRED COMPONENTS Runtime
  PATHS "${_prefix}/lib/cmake/SouiRuntime" NO_DEFAULT_PATH)
find_package(SouiDesktop 0.6.0 EXACT CONFIG REQUIRED COMPONENTS Desktop
  PATHS "${_prefix}/lib/cmake/SouiDesktop" NO_DEFAULT_PATH)
foreach(_package SouiRuntime SouiDesktop)
  file(REAL_PATH "${${_package}_DIR}" _config_directory)
  if(NOT _config_directory STREQUAL "${_prefix}/lib/cmake/${_package}")
    message(FATAL_ERROR "${_package} did not come from SDK_PREFIX")
  endif()
endforeach()

function(require_installed_target target output_archive)
  get_target_property(_includes "${target}" INTERFACE_INCLUDE_DIRECTORIES)
  list(LENGTH _includes _include_count)
  if(NOT _include_count EQUAL 1)
    message(FATAL_ERROR "${target} must expose only the installed include directory")
  endif()
  file(REAL_PATH "${_includes}" _include_real)
  if(NOT _include_real STREQUAL _installed_include)
    message(FATAL_ERROR "${target} includes are outside SDK_PREFIX")
  endif()
  # Read the actual export instead of assuming a platform archive name/suffix.
  get_target_property(_archive "${target}" IMPORTED_LOCATION_RELEASE)
  if(NOT _archive OR NOT IS_ABSOLUTE "${_archive}" OR
     NOT EXISTS "${_archive}" OR IS_DIRECTORY "${_archive}")
    message(FATAL_ERROR "${target} requires an installed Release archive")
  endif()
  file(REAL_PATH "${_archive}" _archive_real)
  cmake_path(IS_PREFIX _prefix "${_archive_real}" NORMALIZE _inside_prefix)
  if(NOT _inside_prefix)
    message(FATAL_ERROR "${target} archive is outside SDK_PREFIX")
  endif()
  set("${output_archive}" "${_archive_real}" PARENT_SCOPE)
endfunction()
require_installed_target(Soui::Runtime _runtime_archive)
require_installed_target(Soui::Desktop _desktop_archive)
set(_loader_archive "not applicable")
if(WIN32)
  if(NOT SouiDesktop_BACKEND STREQUAL "webview2")
    message(FATAL_ERROR "Windows CI requires the installed WebView2 Desktop package")
  endif()
  get_target_property(_loader_archive Soui::DesktopWebView2Loader IMPORTED_LOCATION)
  file(REAL_PATH "${_loader_archive}" _loader_real)
  if(NOT EXISTS "${_loader_archive}" OR
     NOT _loader_real STREQUAL "${_prefix}/lib/SouiDesktop/webview2/WebView2LoaderStatic.lib")
    message(FATAL_ERROR "WebView2 loader is outside the isolated installed package")
  endif()
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package-origin.txt"
  "Runtime=${_runtime_archive}\nDesktop=${_desktop_archive}\nLoader=${_loader_archive}\nincludes=${_installed_include}\n")

add_executable(ci_runtime_consumer runtime.cpp)
target_link_libraries(ci_runtime_consumer PRIVATE Soui::Runtime)
add_executable(ci_desktop_consumer desktop.cpp)
target_link_libraries(ci_desktop_consumer PRIVATE Soui::Desktop)

# Compile every installed public header, including future additions, using only
# the imported include directory. The generated file belongs to this build tree.
foreach(_required soui/diagnostics/runtime_metrics.h
                  soui/diagnostics/view_observation_abi.h soui/desktop/app.h)
  if(NOT EXISTS "${_installed_include}/${_required}")
    message(FATAL_ERROR "Missing installed public header: ${_required}")
  endif()
endforeach()
file(GLOB_RECURSE _public_headers LIST_DIRECTORIES FALSE
  RELATIVE "${_installed_include}" "${_installed_include}/soui/*.h")
list(SORT _public_headers)
set(_header_source "// Generated installed-public-header compilation.\n")
foreach(_header IN LISTS _public_headers)
  if(NOT _header MATCHES "^soui/[A-Za-z0-9_./-]+[.]h$")
    message(FATAL_ERROR "Unsupported installed header path")
  endif()
  string(APPEND _header_source "#include <${_header}>\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/all-installed-headers.cpp" "${_header_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/installed-headers.txt" "${_public_headers}\n")
list(LENGTH _public_headers _header_count)
message(STATUS "CI will compile ${_header_count} installed public headers")
add_library(ci_installed_headers OBJECT "${CMAKE_CURRENT_BINARY_DIR}/all-installed-headers.cpp")
target_link_libraries(ci_installed_headers PRIVATE Soui::Desktop)
foreach(_target ci_runtime_consumer ci_desktop_consumer ci_installed_headers)
  target_compile_features("${_target}" PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options("${_target}" PRIVATE /utf-8 /W4)
    set_property(TARGET "${_target}" PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  else()
    target_compile_options("${_target}" PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

  enable_testing()
