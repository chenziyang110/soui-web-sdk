cmake_minimum_required(VERSION 3.24)
project(SouiInputUiHarnessPolicy LANGUAGES CXX)
add_executable(InputUiHarnessPolicyTests host-options.cpp)
target_compile_features(InputUiHarnessPolicyTests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(InputUiHarnessPolicyTests PRIVATE /W4 /utf-8)
  set_property(TARGET InputUiHarnessPolicyTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
enable_testing()
add_test(NAME benchmark.input-ui-harness-policy COMMAND InputUiHarnessPolicyTests)
set_tests_properties(benchmark.input-ui-harness-policy PROPERTIES TIMEOUT 10)
