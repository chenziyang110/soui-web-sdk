cmake_minimum_required(VERSION 3.24)
project(SouiCacheCalibration LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The SOUI benchmark host requires Windows and MSVC")
endif()
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED)
add_executable(SouiCacheCalibration main.cpp)
target_link_libraries(SouiCacheCalibration PRIVATE Soui::Sdk Soui::ActionJson psapi)
target_compile_features(SouiCacheCalibration PRIVATE cxx_std_17)
target_compile_options(SouiCacheCalibration PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiCacheCalibration PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiCacheCalibration PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
