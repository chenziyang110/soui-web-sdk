cmake_minimum_required(VERSION 3.24)
project(SouiManualProfileStartTests LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The hidden native-menu fixture requires Windows/MSVC-compatible tools")
endif()
add_executable(soui_manual_profile_start_tests fixture.cpp)
target_compile_features(soui_manual_profile_start_tests PRIVATE cxx_std_17)
target_compile_definitions(soui_manual_profile_start_tests PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_compile_options(soui_manual_profile_start_tests PRIVATE /utf-8 /W4 /U_DEBUG)
target_link_libraries(soui_manual_profile_start_tests PRIVATE user32)
set_property(TARGET soui_manual_profile_start_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
enable_testing()
add_test(NAME performance.native.manual-profile-start COMMAND soui_manual_profile_start_tests)
