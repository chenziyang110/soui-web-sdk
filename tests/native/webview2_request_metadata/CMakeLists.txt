cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiWebView2RequestMetadata LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(metadata_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(soui_webview2_request_metadata_tests fixture.cpp)
target_include_directories(soui_webview2_request_metadata_tests PRIVATE "${metadata_root}/third_party/include/json")
target_compile_features(soui_webview2_request_metadata_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(soui_webview2_request_metadata_tests PRIVATE /EHsc /utf-8 /W4 /U_DEBUG /UNDEBUG)
  set_property(TARGET soui_webview2_request_metadata_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
add_test(NAME sdk.native.webview2-request-metadata COMMAND soui_webview2_request_metadata_tests)
set_tests_properties(sdk.native.webview2-request-metadata PROPERTIES TIMEOUT 15)
option(SOUI_METADATA_MICROBENCHMARK "Build opt-in CPU-only request metadata comparison" OFF)
if(SOUI_METADATA_MICROBENCHMARK)
  add_executable(soui_webview2_request_metadata_benchmark benchmark.cpp)
  target_include_directories(soui_webview2_request_metadata_benchmark PRIVATE "${metadata_root}/third_party/include/json")
  target_compile_features(soui_webview2_request_metadata_benchmark PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(soui_webview2_request_metadata_benchmark PRIVATE /EHsc /utf-8 /W4)
    set_property(TARGET soui_webview2_request_metadata_benchmark PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  endif()
endif()
