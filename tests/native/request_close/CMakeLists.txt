# Use the already validated lifetime generation contract: exact HEX extraction,
# LF-only/NUL-free source with terminal LF, escaped dollar bytes and UNIX output.
# Build-before checks compare generated bytes against the unchanged actual input.
function(close_extract name source begin_marker end_marker)
  file(READ "${source}" source_hex HEX)
  string(TOLOWER "${source_hex}" source_hex)
  string(HEX "${begin_marker}" begin_hex)
  string(HEX "${end_marker}" end_hex)
  foreach(marker IN ITEMS begin end)
    string(REGEX MATCHALL "${${marker}_hex}" matches "${source_hex}")
    list(LENGTH matches count)
    if(NOT count EQUAL 1)
      message(FATAL_ERROR "Request-close ${name} requires one unique ${marker} marker")
    endif()
  endforeach()
  string(FIND "${source_hex}" "${begin_hex}" begin_offset)
  string(FIND "${source_hex}" "${end_hex}" end_offset)
  math(EXPR begin_alignment "${begin_offset} % 2")
  math(EXPR end_alignment "${end_offset} % 2")
  if(NOT begin_alignment EQUAL 0 OR NOT end_alignment EQUAL 0 OR end_offset LESS_EQUAL begin_offset)
    message(FATAL_ERROR "Request-close ${name} has invalid byte boundaries")
  endif()
  math(EXPR fragment_length "${end_offset} - ${begin_offset}")
  string(SUBSTRING "${source_hex}" ${begin_offset} ${fragment_length} fragment_hex)
  math(EXPR last_offset "${fragment_length} - 2")
  string(SUBSTRING "${fragment_hex}" ${last_offset} 2 terminal_hex)
  if(NOT terminal_hex STREQUAL "0a")
    message(FATAL_ERROR "Request-close ${name} requires a terminal LF; missing final LF is unsupported")
  endif()
  set(fragment "")
  foreach(offset RANGE 0 ${last_offset} 2)
    string(SUBSTRING "${fragment_hex}" ${offset} 2 byte_hex)
    math(EXPR byte "0x${byte_hex}")
    if(byte EQUAL 0 OR byte EQUAL 13)
      message(FATAL_ERROR "Request-close ${name} requires NUL-free LF-only bytes; CRLF/mixed input is unsupported")
    endif()
    string(ASCII ${byte} character)
    string(APPEND fragment "${character}")
  endforeach()
  set(generated "${close_generated_directory}/${name}.inc")
  string(REPLACE "$" "$<1:$>" protected_content "${fragment}")
  file(GENERATE OUTPUT "${generated}" CONTENT "${protected_content}" NEWLINE_STYLE UNIX)
  file(SHA256 "${source}" source_sha)
  string(SHA256 fragment_sha "${fragment}")
  message(STATUS "Request-close ${name} expected actual fragment SHA256: ${fragment_sha}")
  list(APPEND close_byte_commands
    COMMAND "${CMAKE_COMMAND}" "-DCLOSE_SOURCE=${source}" "-DCLOSE_GENERATED=${generated}"
    "-DCLOSE_SOURCE_SHA=${source_sha}" "-DCLOSE_BEGIN_HEX_OFFSET=${begin_offset}"
    "-DCLOSE_FRAGMENT_HEX_LENGTH=${fragment_length}" -P "${close_generated_directory}/check_source_bytes.cmake")
  set(close_byte_commands "${close_byte_commands}" PARENT_SCOPE)
endfunction()

cmake_minimum_required(VERSION 3.24)
foreach(required IN ITEMS CLOSE_SOURCE CLOSE_GENERATED CLOSE_SOURCE_SHA CLOSE_BEGIN_HEX_OFFSET CLOSE_FRAGMENT_HEX_LENGTH)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Request-close byte gate argument is missing: ${required}")
  endif()

if(NOT IS_ABSOLUTE "${CLOSE_SOURCE}" OR NOT IS_ABSOLUTE "${CLOSE_GENERATED}"
    OR NOT CLOSE_SOURCE_SHA MATCHES "^[a-f0-9]+$"
    OR NOT CLOSE_BEGIN_HEX_OFFSET MATCHES "^[0-9]+$"
    OR NOT CLOSE_FRAGMENT_HEX_LENGTH MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Request-close byte gate arguments are invalid")
endif()
string(LENGTH "${CLOSE_SOURCE_SHA}" hash_length)
if(NOT hash_length EQUAL 64)
  message(FATAL_ERROR "Request-close source hash has invalid length")
endif()
file(SHA256 "${CLOSE_SOURCE}" current_sha)
if(NOT current_sha STREQUAL CLOSE_SOURCE_SHA)
  message(FATAL_ERROR "Request-close actual source changed after configuration")
endif()
file(READ "${CLOSE_SOURCE}" source_hex HEX)
string(TOLOWER "${source_hex}" source_hex)
string(SUBSTRING "${source_hex}" ${CLOSE_BEGIN_HEX_OFFSET} ${CLOSE_FRAGMENT_HEX_LENGTH} expected_hex)
file(READ "${CLOSE_GENERATED}" generated_hex HEX)
string(TOLOWER "${generated_hex}" generated_hex)
if(NOT generated_hex STREQUAL expected_hex)
  message(FATAL_ERROR "Request-close generated include differs from actual source bytes")
endif()
]==])
