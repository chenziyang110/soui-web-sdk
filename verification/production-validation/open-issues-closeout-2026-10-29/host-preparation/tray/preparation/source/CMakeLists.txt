cmake_minimum_required(VERSION 3.24)
project(SouiTrayConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Tray consumer requires the supported Windows/MSVC SDK toolchain")
endif()
find_package(SouiSdk 0.6.0 EXACT CONFIG REQUIRED)
add_executable(SouiTrayConsumer main.cpp)
target_link_libraries(SouiTrayConsumer PRIVATE Soui::Sdk)
target_compile_options(SouiTrayConsumer PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiTrayConsumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiTrayConsumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
