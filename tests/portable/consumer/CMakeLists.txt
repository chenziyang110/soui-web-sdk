cmake_minimum_required(VERSION 3.24)
project(SouiRuntimeConsumer LANGUAGES CXX)
find_package(SouiRuntime CONFIG REQUIRED COMPONENTS Runtime)
add_executable(runtime_consumer main.cpp)
target_link_libraries(runtime_consumer PRIVATE Soui::Runtime)
if(MSVC)
  target_compile_options(runtime_consumer PRIVATE /utf-8)
  set_property(TARGET runtime_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
enable_testing()
add_test(NAME sdk.portable.installed-consumer COMMAND runtime_consumer)
