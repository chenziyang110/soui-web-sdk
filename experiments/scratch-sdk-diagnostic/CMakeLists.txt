cmake_minimum_required(VERSION 3.24)
project(SouiScratchDiagnostic LANGUAGES C CXX)
get_filename_component(DIAGNOSTIC_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SOUI_BUILD_WEBVIEW2 ON CACHE BOOL "Diagnostic modern backend" FORCE)
add_subdirectory("${DIAGNOSTIC_SDK_ROOT}/sdk" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
set(recipe_root "${CMAKE_CURRENT_SOURCE_DIR}/../scratch-webview2-baseline")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${recipe_root}/diagnostic-protocol.json" "${recipe_root}/diagnostic-page-scripts.json" "${recipe_root}/diagnostic-probe.js")
file(READ "${recipe_root}/diagnostic-protocol.json" DIAGNOSTIC_PROTOCOL)
file(READ "${recipe_root}/diagnostic-page-scripts.json" DIAGNOSTIC_SCRIPTS)
file(READ "${recipe_root}/diagnostic-probe.js" DIAGNOSTIC_PROBE)
foreach(input IN ITEMS DIAGNOSTIC_PROTOCOL DIAGNOSTIC_SCRIPTS DIAGNOSTIC_PROBE)
  string(FIND "${${input}}" ")SOUIDIAG\"" delimiter)
  if(NOT delimiter EQUAL -1)
    message(FATAL_ERROR "Diagnostic input contains its C++ literal delimiter")
  endif()
endforeach()
configure_file(recipe.h.in "${CMAKE_CURRENT_BINARY_DIR}/diagnostic_recipe.h" @ONLY)
# The production implementation is compiled unchanged in a dedicated test TU.
# Neither this extension nor its private commands enter Soui::WebView2.
add_library(ScratchDiagnosticHost STATIC sdk-access.cpp)
get_target_property(renderer_includes soui_webview2 INCLUDE_DIRECTORIES)
get_target_property(renderer_links soui_webview2 INTERFACE_LINK_LIBRARIES)
target_include_directories(ScratchDiagnosticHost PRIVATE ${renderer_includes} "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ScratchDiagnosticHost PUBLIC ${renderer_links} Soui::ActionJson)
target_compile_definitions(ScratchDiagnosticHost PUBLIC SOUI_HOST_STAGE0_DEVTOOLS_TEST_ONLY SOUI_HOST_PERFORMANCE_PROFILE_TEST_ONLY)
target_compile_features(ScratchDiagnosticHost PUBLIC cxx_std_17)
target_compile_options(ScratchDiagnosticHost PRIVATE /utf-8 /W4 /U_DEBUG /EHsc)
set_property(TARGET ScratchDiagnosticHost PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(SouiScratchDiagnostic WIN32 main.cpp ../../demos/scratch/app/application.manifest)
target_include_directories(SouiScratchDiagnostic PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(SouiScratchDiagnostic PRIVATE ScratchDiagnosticHost)
target_compile_options(SouiScratchDiagnostic PRIVATE /utf-8 /W4 /U_DEBUG /EHsc)
set_target_properties(SouiScratchDiagnostic PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
