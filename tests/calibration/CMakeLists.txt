cmake_minimum_required(VERSION 3.24)
project(SouiTaskStoreCalibration LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Calibration requires the supported Windows SDK toolchain")
endif()
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED)
add_executable(SouiTaskStoreCalibration task-store-quota.cpp)
target_link_libraries(SouiTaskStoreCalibration PRIVATE Soui::Sdk Soui::ActionJson psapi)
target_compile_features(SouiTaskStoreCalibration PRIVATE cxx_std_17)
target_compile_options(SouiTaskStoreCalibration PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiTaskStoreCalibration PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_options(SouiTaskStoreCalibration PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/SouiTaskStoreCalibration.map")
