cmake_minimum_required(VERSION 3.24)
project(SouiEndpointConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The installed endpoint consumer requires Windows/MSVC")
endif()
find_package(SouiSdk 0.5.1 EXACT CONFIG REQUIRED)
add_executable(SouiEndpointConsumer main.cpp)
target_link_libraries(SouiEndpointConsumer PRIVATE Soui::Sdk)
target_compile_options(SouiEndpointConsumer PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiEndpointConsumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
