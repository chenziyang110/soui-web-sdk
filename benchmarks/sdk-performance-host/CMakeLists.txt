# The package config itself is an input too. Pin every imported library and
# include path used by this consumer to the exact frozen prefix; the Node
# verifier then proves the matching bytes against the build receipt/archive.
string(TOUPPER "${MATCHED_SDK_CONFIGURATION}" _sdk_config_upper)
function(_soui_perf_assert_imported_location target expected_relative)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Required installed SDK target is missing: ${target}")
  endif()
  get_target_property(_imported "${target}" IMPORTED)
  if(NOT _imported)
    message(FATAL_ERROR "SDK package target is not imported: ${target}")
  endif()
  get_target_property(_locations "${target}" IMPORTED_CONFIGURATIONS)
  get_target_property(_actual "${target}" "IMPORTED_LOCATION_${_sdk_config_upper}")
  if(NOT _actual AND NOT _locations)
    get_target_property(_actual "${target}" IMPORTED_LOCATION)
  endif()
  if(NOT _actual OR _actual MATCHES "\\$<")
    message(FATAL_ERROR "SDK target has no concrete ${MATCHED_SDK_CONFIGURATION} location: ${target}")
  endif()
  file(REAL_PATH "${_actual}" _actual_real)
  file(REAL_PATH "${_requested_sdk}/${expected_relative}" _expected_real)
  string(TOLOWER "${_actual_real}" _actual_compare)
  string(TOLOWER "${_expected_real}" _expected_compare)
  if(NOT _actual_compare STREQUAL _expected_compare)
    message(FATAL_ERROR "SDK imported target escapes or differs from frozen prefix: ${target}")
  endif()
endfunction()

function(_soui_perf_assert_imported_includes target expected_relative)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Required installed SDK target is missing: ${target}")
  endif()
  get_target_property(_includes "${target}" INTERFACE_INCLUDE_DIRECTORIES)
  if(NOT _includes)
    message(FATAL_ERROR "SDK target has no installed include directory: ${target}")
  endif()
  file(REAL_PATH "${_requested_sdk}/${expected_relative}" _expected_real)
  string(TOLOWER "${_expected_real}" _expected_compare)
  set(_expected_found FALSE)
  foreach(_include IN LISTS _includes)
    if(NOT IS_ABSOLUTE "${_include}" OR "${_include}" MATCHES "\\$<")
      message(FATAL_ERROR "SDK imported include path is not concrete: ${target}")
    endif()
    file(REAL_PATH "${_include}" _include_real)
    cmake_path(IS_PREFIX _requested_sdk "${_include_real}" NORMALIZE _within_prefix)
    if(NOT _within_prefix)
      message(FATAL_ERROR "SDK imported include path escapes frozen prefix: ${target}")
    endif()
    string(TOLOWER "${_include_real}" _include_compare)
    if(_include_compare STREQUAL _expected_compare)
      set(_expected_found TRUE)
    endif()
  endforeach()
  if(NOT _expected_found)
    message(FATAL_ERROR "SDK imported include path differs from frozen prefix: ${target}")
  endif()
endfunction()

_soui_perf_assert_imported_location(Soui::Sdk "lib/${MATCHED_SDK_CONFIGURATION}/soui_sdk.lib")
_soui_perf_assert_imported_location(Soui::WebView2 "lib/${MATCHED_SDK_CONFIGURATION}/soui_webview2.lib")
_soui_perf_assert_imported_location(Soui::WebView2Loader "lib/webview2/WebView2LoaderStatic.lib")
_soui_perf_assert_imported_includes(Soui::Sdk "include")
_soui_perf_assert_imported_includes(Soui::ActionJson "include/soui/action-json")
get_target_property(_webview2_links Soui::WebView2 INTERFACE_LINK_LIBRARIES)
foreach(_required_link IN ITEMS Soui::Sdk Soui::WebView2Loader)
  list(FIND _webview2_links "${_required_link}" _required_link_index)
  if(_required_link_index EQUAL -1)
    message(FATAL_ERROR "Soui::WebView2 no longer links its frozen SDK dependency ${_required_link}")
  endif()
endforeach()

set(_sdk_input_manifest "${CMAKE_CURRENT_BINARY_DIR}/sdk-build-inputs.json")
set(_sdk_identity_header "${CMAKE_CURRENT_BINARY_DIR}/frozen_sdk_identity.h")
set(_sdk_identity_arguments "${_requested_sdk}" "${_sdk_build_receipt}" "${MATCHED_WEBVIEW2_SDK_ARCHIVE}" "${MATCHED_SDK_VERSION}"
  "${_sdk_architecture}" "${MATCHED_SDK_CONFIGURATION}" "${_sdk_input_manifest}")
execute_process(COMMAND "${SOUI_PERF_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/freeze-sdk-build.cjs"
  freeze ${_sdk_identity_arguments} "${_sdk_identity_header}"
  RESULT_VARIABLE _sdk_identity_status OUTPUT_QUIET ERROR_VARIABLE _sdk_identity_error)
if(NOT _sdk_identity_status EQUAL 0)
  message(FATAL_ERROR "Installed SDK source binding failed: ${_sdk_identity_error}")
endif()
add_custom_target(soui_perf_verify_sdk_inputs
  COMMAND "${SOUI_PERF_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/freeze-sdk-build.cjs" verify
    "${_requested_sdk}" "${_sdk_build_receipt}" "${MATCHED_WEBVIEW2_SDK_ARCHIVE}" "${MATCHED_SDK_VERSION}" "${_sdk_architecture}"
    "$<CONFIG>" "${_sdk_input_manifest}" VERBATIM)
