cmake_minimum_required(VERSION 3.24)
project(ReferenceGuiController LANGUAGES CXX)
find_package(SouiSdk 0.5.0 EXACT CONFIG REQUIRED)
if(NOT DEFINED REFERENCE_SOURCE_ROOT)
  message(FATAL_ERROR "Explicit current reference source required")
endif()
add_executable(ReferenceGuiController controller.cpp)
target_include_directories(ReferenceGuiController PRIVATE "${REFERENCE_SOURCE_ROOT}/tests")
target_link_libraries(ReferenceGuiController PRIVATE Soui::Sdk Soui::ActionJson user32)
target_compile_features(ReferenceGuiController PRIVATE cxx_std_17)
target_compile_options(ReferenceGuiController PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET ReferenceGuiController PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_options(ReferenceGuiController PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/ReferenceGuiController.map")
