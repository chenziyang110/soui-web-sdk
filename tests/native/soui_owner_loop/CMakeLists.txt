cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiOwnerLoopConsumer LANGUAGES CXX)
endif()
if(NOT WIN32 OR NOT (MSVC OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC"))
  message(FATAL_ERROR "The SOUI consumer requires the Windows MSVC-front-end toolchain")
endif()
get_filename_component(_repository_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${_repository_root}/benchmarks/input-ui-host/sdk_version.cmake")
set(SOUI_OWNER_LOOP_SDK_ROOT "" CACHE PATH "Independently installed SDK prefix for standalone use")
set(SOUI_OWNER_LOOP_SDK_VERSION "" CACHE STRING "Exact raw installed SDK release for standalone use")
if(NOT TARGET Soui::Sdk)
  if(NOT IS_ABSOLUTE "${SOUI_OWNER_LOOP_SDK_ROOT}" OR "${SOUI_OWNER_LOOP_SDK_VERSION}" STREQUAL "")
    message(FATAL_ERROR "Standalone mode requires explicit SOUI_OWNER_LOOP_SDK_ROOT and SOUI_OWNER_LOOP_SDK_VERSION")
  endif()
  soui_input_ui_package_version("${SOUI_OWNER_LOOP_SDK_VERSION}" _package_version)
  find_package(SouiSdk ${_package_version} EXACT CONFIG REQUIRED
    PATHS "${SOUI_OWNER_LOOP_SDK_ROOT}/lib/cmake/SouiSdk" NO_DEFAULT_PATH)
  file(REAL_PATH "${SOUI_OWNER_LOOP_SDK_ROOT}" _requested_prefix)
  file(REAL_PATH "${SouiSdk_DIR}/../../.." _actual_prefix)
  if(NOT _requested_prefix STREQUAL _actual_prefix)
    message(FATAL_ERROR "Resolved SDK prefix differs")
  endif()
  get_target_property(_imported Soui::Sdk IMPORTED)
  if(NOT _imported)
    message(FATAL_ERROR "Standalone mode requires the normal installed SDK")
  endif()
  set(SOUI_INPUT_UI_SDK_VERSION "${SOUI_OWNER_LOOP_SDK_VERSION}")
endif()
soui_input_ui_resolve_sdk()
get_filename_component(_sdk_header_directory "${SOUI_INPUT_UI_SDK_VERSION_HEADER}" DIRECTORY)
if(NOT EXISTS "${_sdk_header_directory}/host/soui_owner_application.h")
  message(FATAL_ERROR "The actual SDK must provide public soui_owner_application.h")
endif()
set(SOUI_OWNER_LOOP_DEPENDENCY_ROOT "${_repository_root}/third_party" CACHE PATH "Existing SOUI dependency headers/libs/DLLs")
if(NOT IS_ABSOLUTE "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}")
  message(FATAL_ERROR "SOUI dependency root must be an explicit actual directory")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(_soui_arch x64)
else()
  set(_soui_arch x86)
endif()
set(_soui_bin "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/prebuilt/${_soui_arch}/soui/bin")
if(NOT EXISTS "${_soui_bin}/render-gdi.dll")
  message(FATAL_ERROR "Existing normal render-gdi dependency is required; this target does not rebuild vendors")
endif()
add_executable(soui_owner_loop_consumer consumer.cpp)
target_compile_features(soui_owner_loop_consumer PRIVATE cxx_std_17)
target_compile_options(soui_owner_loop_consumer PRIVATE /utf-8 /W4 /EHsc /U_DEBUG)
target_compile_definitions(soui_owner_loop_consumer PRIVATE UNICODE _UNICODE NOMINMAX NDEBUG
  WINVER=0x0601 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)
set_property(TARGET soui_owner_loop_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_include_directories(soui_owner_loop_consumer PRIVATE
  "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/include/soui/config"
  "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/include/soui/SOUI/include"
  "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/include/soui/utilities/include"
  "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/include/soui/components")
target_link_directories(soui_owner_loop_consumer PRIVATE "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/prebuilt/${_soui_arch}/soui/lib")
target_link_libraries(soui_owner_loop_consumer PRIVATE Soui::Sdk soui utilities shlwapi)
file(GLOB _soui_dlls "${_soui_bin}/*.dll")
list(FILTER _soui_dlls EXCLUDE REGEX "(souid|utilitiesd|render-gdid|imgdecoder-gdipd|soui-sys-resourced)\\.dll$")
file(GLOB_RECURSE _soui_crt "${SOUI_OWNER_LOOP_DEPENDENCY_ROOT}/prebuilt/${_soui_arch}/runtime/*.dll")
add_custom_command(TARGET soui_owner_loop_consumer POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
  ${_soui_dlls} ${_soui_crt} "$<TARGET_FILE_DIR:soui_owner_loop_consumer>" VERBATIM)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sdk-identity-$<CONFIG>.txt"
  CONTENT "raw=${SOUI_INPUT_UI_EFFECTIVE_SDK_VERSION}\npackage=${SOUI_INPUT_UI_PACKAGE_VERSION}\nheader=${SOUI_INPUT_UI_SDK_VERSION_HEADER}\nsdk=$<TARGET_FILE:Soui::Sdk>\ndependency_arch=${_soui_arch}\n")
enable_testing()
add_test(NAME sdk.native.soui-owner-loop.consumer COMMAND soui_owner_loop_consumer)
set_tests_properties(sdk.native.soui-owner-loop.consumer PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
