cmake_minimum_required(VERSION 3.24)
project(SouiAcceptance LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Acceptance requires the supported Windows/MSVC SDK toolchain")
endif()
find_package(SouiSdk 0.3.0 EXACT CONFIG REQUIRED)
add_executable(SouiAcceptance main.cpp)
target_link_libraries(SouiAcceptance PRIVATE Soui::Sdk psapi)
target_compile_options(SouiAcceptance PRIVATE /W4 /utf-8 /U_DEBUG)
set_property(TARGET SouiAcceptance PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(SouiAcceptance PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
foreach(ACCEPTANCE_WINDOW_ID RANGE 1 3)
  configure_file(page.html.in "resources/window-${ACCEPTANCE_WINDOW_ID}/index.html" @ONLY)
endforeach()
