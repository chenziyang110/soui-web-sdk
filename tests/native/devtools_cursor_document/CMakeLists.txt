cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(InspectorCursorDocumentRegression LANGUAGES CXX)
  enable_testing()
endif()

get_filename_component(sdk_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(broker "${sdk_root}/third_party/sources/miniblink/wke/CursorDescriptorBroker.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${broker}")
file(READ "${broker}" broker_source)
string(FIND "${broker_source}" "void brokerDocumentCommitted(wkeWebView view, bool mainDocument)\n{" start)
string(FIND "${broker_source}" "\nBrokerNativeScope::BrokerNativeScope(" end)
if(start LESS 0 OR end LESS start)
  message(FATAL_ERROR "The production brokerDocumentCommitted extraction boundary changed")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${broker_source}" ${start} ${length} implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/broker-document.inc" "${implementation}\n")

# Compile the actual production transition with finite external dependencies.
add_executable(soui_devtools_cursor_document_tests cursor_document_tests.cpp)
target_include_directories(soui_devtools_cursor_document_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(soui_devtools_cursor_document_tests PRIVATE cxx_std_17)
target_compile_options(soui_devtools_cursor_document_tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME sdk.native.devtools-cursor-document COMMAND soui_devtools_cursor_document_tests)
set_tests_properties(sdk.native.devtools-cursor-document PROPERTIES TIMEOUT 10)
