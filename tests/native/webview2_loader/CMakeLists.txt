cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SouiOptionalLoaderTests LANGUAGES CXX)
  get_filename_component(SOUI_DISTRIBUTION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
  enable_testing()
endif()
find_path(SOUI_WEBVIEW2_INCLUDE WebView2.h PATHS "${SOUI_WEBVIEW2_SDK_ROOT}/build/native/include" NO_DEFAULT_PATH REQUIRED)
foreach(mode IN ITEMS complete version-only create-only)
  add_library(loader_fixture_${mode} SHARED fixture.cpp)
  target_include_directories(loader_fixture_${mode} PRIVATE "${SOUI_WEBVIEW2_INCLUDE}")
  target_compile_features(loader_fixture_${mode} PRIVATE cxx_std_17)
  target_compile_options(loader_fixture_${mode} PRIVATE /W4 /utf-8 /U_DEBUG)
  set_property(TARGET loader_fixture_${mode} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  set_target_properties(loader_fixture_${mode} PROPERTIES OUTPUT_NAME WebView2Loader
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${mode}/$<CONFIG>"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${mode}/$<CONFIG>")
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(version_export loader_version)
    set(create_export loader_environment)
  else()
    set(version_export "_loader_version@8")
    set(create_export "_loader_environment@16")
  endif()
  set(exports "EXPORTS\n")
  if(NOT mode STREQUAL "create-only")
    string(APPEND exports "  GetAvailableCoreWebView2BrowserVersionString=${version_export}\n")
  endif()
  if(NOT mode STREQUAL "version-only")
    string(APPEND exports "  CreateCoreWebView2EnvironmentWithOptions=${create_export}\n")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${mode}.def" "${exports}")
  target_sources(loader_fixture_${mode} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${mode}.def")
  target_link_libraries(loader_fixture_${mode} PRIVATE ole32)
endforeach()
foreach(mode IN ITEMS missing version-only create-only complete invalid-image locked)
  add_executable(loader_test_${mode} loader_tests.cpp
    "${SOUI_DISTRIBUTION_ROOT}/sdk/renderers/webview2/webview2_loader_bridge.cpp")
  target_compile_features(loader_test_${mode} PRIVATE cxx_std_17)
  target_compile_options(loader_test_${mode} PRIVATE /W4 /utf-8 /U_DEBUG /EHsc)
  set_property(TARGET loader_test_${mode} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_include_directories(loader_test_${mode} PRIVATE "${SOUI_WEBVIEW2_INCLUDE}"
    "${SOUI_DISTRIBUTION_ROOT}/sdk/include" "${SOUI_DISTRIBUTION_ROOT}/sdk/renderers/webview2")
  target_link_libraries(loader_test_${mode} PRIVATE ole32)
  add_dependencies(loader_test_${mode} loader_fixture_complete)
  set_target_properties(loader_test_${mode} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cases/${mode}/$<CONFIG>")
  if(mode MATCHES "^(complete|version-only|create-only)$")
    add_custom_command(TARGET loader_test_${mode} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "$<TARGET_FILE:loader_fixture_${mode}>" "$<TARGET_FILE_DIR:loader_test_${mode}>/WebView2Loader.dll")
  endif()
  add_test(NAME sdk.native.webview2-loader-${mode} COMMAND loader_test_${mode}
    "${mode}" "$<TARGET_FILE:loader_fixture_complete>")
  set_tests_properties(sdk.native.webview2-loader-${mode} PROPERTIES TIMEOUT 10)
  if(mode STREQUAL "missing")
    set_tests_properties(sdk.native.webview2-loader-${mode} PROPERTIES
      WORKING_DIRECTORY "$<TARGET_FILE_DIR:loader_fixture_complete>")
  endif()
endforeach()
