cmake_minimum_required(VERSION 3.24)
project(SouiR42ColdGuardInput LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use the frozen standard Windows MSVC toolchain")
endif()
set(CMAKE_SUPPRESS_REGENERATION ON)
set(R42_JSON_INCLUDE "" CACHE PATH "Exact frozen installed ActionJson header directory")
if(NOT IS_ABSOLUTE "${R42_JSON_INCLUDE}" OR NOT EXISTS "${R42_JSON_INCLUDE}/json.hpp")
  message(FATAL_ERROR "Freeze the existing installed ActionJson header explicitly")
endif()
add_executable(R42NativeStripInput input.cpp)
target_include_directories(R42NativeStripInput PRIVATE "${R42_JSON_INCLUDE}")
target_compile_features(R42NativeStripInput PRIVATE cxx_std_17)
target_compile_options(R42NativeStripInput PRIVATE /W4 /EHsc /utf-8 /U_DEBUG)
target_compile_definitions(R42NativeStripInput PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
set_property(TARGET R42NativeStripInput PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(R42NativeStripInput PRIVATE user32 gdi32)
